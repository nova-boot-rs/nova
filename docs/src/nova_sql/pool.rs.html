1use crate::connection::NovaSql;
2use sea_orm::DatabaseConnection;
3use std::sync::{
4    Arc,
5    atomic::{AtomicUsize, Ordering},
6};
7use tokio::sync::RwLock;
8
9/// Simple read/write pool with round-robin replica selection for reads.
10///
11/// This pool exposes `read()` for selecting a replica (or primary when no
12/// replicas exist) and `write()` for obtaining the primary connection for
13/// write operations. It is intentionally small and clones `DatabaseConnection`
14/// references for simplicity.
15#[derive(Clone)]
16pub struct ReadWritePool {
17    primary: DatabaseConnection,
18    replicas: Arc<RwLock<Vec<DatabaseConnection>>>,
19    rr: Arc<AtomicUsize>,
20}
21
22impl ReadWritePool {
23    pub fn new(
24        primary: DatabaseConnection,
25        replicas: Arc<RwLock<Vec<DatabaseConnection>>>,
26    ) -> Self {
27        Self {
28            primary,
29            replicas,
30            rr: Arc::new(AtomicUsize::new(0)),
31        }
32    }
33
34    /// Choose a replica connection for read queries. Async because replicas list is protected by an async lock.
35    pub async fn read(&self) -> DatabaseConnection {
36        let reps = self.replicas.read().await;
37        if reps.is_empty() {
38            return self.primary.clone();
39        }
40
41        let idx = self.rr.fetch_add(1, Ordering::Relaxed);
42        reps[idx % reps.len()].clone()
43    }
44
45    /// Return the primary connection for writes.
46    pub fn write(&self) -> DatabaseConnection {
47        self.primary.clone()
48    }
49
50    /// Add a replica connection dynamically.
51    pub async fn add_replica(&self, conn: DatabaseConnection) {
52        self.replicas.write().await.push(conn);
53    }
54}
55
56impl NovaSql {
57    /// Construct a `ReadWritePool` for injection into handlers; clones internal references.
58    pub fn read_write_pool(&self) -> ReadWritePool {
59        ReadWritePool::new(self.db.clone(), self.replicas.clone())
60    }
61}
