9/// Axum middleware that resolves tenant from the request.
10pub async fn tenant_middleware(
11    axum::extract::State(resolver): axum::extract::State<Arc<dyn TenantResolver>>,
12    mut request: Request,
13    next: Next,
14) -> Response {
15    let headers = request.headers().clone();
16    let path = request.uri().path().to_string();
17
18    if let Some(tenant) = resolver.resolve(&headers, &path).await {
19        request.extensions_mut().insert(tenant);
20    }
21    next.run(request).await
22}

24/// Convenience extractor for handlers.
25#[derive(Debug, Clone)]
26pub struct CurrentTenant(pub Tenant);
27
28#[async_trait::async_trait]
29impl<S: Send + Sync> FromRequestParts<S> for CurrentTenant {
30    type Rejection = (axum::http::StatusCode, &'static str);
31
32    fn from_request_parts(
33        parts: &mut axum::http::request::Parts,
34        _state: &S,
35    ) -> impl std::future::Future<Output = Result<Self, Self::Rejection>> + Send {
36        let tenant = parts.extensions.get::<Tenant>().cloned().map(CurrentTenant);
37
38        async move { tenant.ok_or((axum::http::StatusCode::UNAUTHORIZED, "Tenant not found")) }
39    }
40}
