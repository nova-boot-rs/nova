1use crate::{connection::NovaSql, pool::ReadWritePool, tenant_middleware::tenant_middleware};
2use nova_core::{NovaPlugin, async_trait, axum::Extension, axum::Router, axum::middleware};
3use std::sync::Arc;
4
5/// Plugin integration for Nova applications.
6///
7/// Implements `NovaPlugin` for `NovaSql` so the plugin can be added to the
8/// application via `NovaApp::add_plugin(...)`. During `extend_router` the
9/// plugin injects a `ReadWritePool` into request extensions so handlers can
10/// extract `NovaDb`.
11#[async_trait]
12impl NovaPlugin for NovaSql {
13    fn name(&self) -> &'static str {
14        "NovaSql (Relational Engine)"
15    }
16
17    async fn on_init(&self) {
18        println!("🗄️ Initializing SQL Plugin...");
19        for task in &self.sync_tasks {
20            task.run(self).await;
21        }
22    }
23
24    fn extend_router(&self, mut router: Router<()>) -> Router<()> {
25        let pool: ReadWritePool = self.read_write_pool();
26        router = router.layer(Extension(pool));
27
28        if let Some(resolver) = &self.tenant_resolver {
29            router = router.layer(middleware::from_fn_with_state(
30                Arc::clone(resolver),
31                tenant_middleware,
32            ));
33        }
34        router
35    }
36}
