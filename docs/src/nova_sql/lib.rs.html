1//! Nova SQL abstractions and relational persistence helpers.
2//!
3//! This crate provides an opinionated integration with `sea-orm` for Nova
4//! applications. It offers:
5//!
6//! - Connection helpers and a `NovaSql` builder for common database options.
7//! - A `ReadWritePool` with round-robin replica selection for read scaling.
8//! - Schema synchronization helpers and migration utilities compatible with
9//!   `sea-orm-migration`.
10//! - An Axum extractor `NovaDb` to access the injected `ReadWritePool` from
11//!   request handlers.
12//!
13//! Examples
14//!
15//! ```rust,ignore
16//! use nova_sql::NovaSql;
17//!
18//! #[tokio::main]
19//! async fn main() -> anyhow::Result<()> {
20//!     let sql = NovaSql::connect("sqlite::memory:", false).await;
21//!     let pool = sql.read_write_pool();
22//!     // Use `pool.read().await` for queries and `pool.write()` for writes.
23//!     Ok(())
24//! }
25//! ```
