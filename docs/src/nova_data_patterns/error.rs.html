3/// Errors returned by CQRS stores and operations.
4#[derive(Debug, Clone)]
5pub enum CqrsError {
6    /// Backend-specific error.
7    Backend(String),
8    /// Serialization/deserialization error.
9    Serialization(String),
10    /// Optimistic concurrency violation.
11    Concurrency(String),
12}
13
14impl fmt::Display for CqrsError {
15    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
16        match self {
17            Self::Backend(msg) => write!(f, "CQRS backend error: {msg}"),
18            Self::Serialization(msg) => write!(f, "CQRS serialization error: {msg}"),
19            Self::Concurrency(msg) => write!(f, "CQRS concurrency error: {msg}"),
20        }
21    }
22}
23
24impl std::error::Error for CqrsError {}
25
26impl From<serde_json::Error> for CqrsError {
27    fn from(err: serde_json::Error) -> Self {
28        Self::Serialization(err.to_string())
29    }
30}

32/// Errors used by the Saga orchestration helpers.
33#[derive(Debug, Clone)]
34pub enum SagaError {
35    /// A saga step failed with a specific reason.
36    StepFailed { step: String, reason: String },
37    /// Compensation for a step failed.
38    CompensationFailed { step: String, reason: String },
39    /// Step timed out.
40    Timeout { step: String },
41    /// Generic abort with message.
42    Aborted(String),
43}
44
45impl fmt::Display for SagaError {
46    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
47        match self {
48            Self::StepFailed { step, reason } => {
49                write!(f, "saga step '{step}' failed: {reason}")
50            }
51            Self::CompensationFailed { step, reason } => {
52                write!(f, "saga compensation for step '{step}' failed: {reason}")
53            }
54            Self::Timeout { step } => {
55                write!(f, "saga step '{step}' timed out")
56            }
57            Self::Aborted(msg) => write!(f, "saga aborted: {msg}"),
58        }
59    }
60}
61
62impl std::error::Error for SagaError {}
