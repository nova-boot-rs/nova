2use async_trait::async_trait;
3use serde_json::Value as JsonValue;
4use std::collections::HashMap;
5use std::sync::Arc;
6use tokio::sync::RwLock;

8/// In-memory adapter useful for tests and local development.
9#[derive(Default)]
10pub struct InMemoryDocumentStore {
11    data: Arc<RwLock<HashMap<String, HashMap<String, JsonValue>>>>,
12    indexes: Arc<RwLock<HashMap<String, Vec<NoSqlIndex>>>>,
13}
14
15/// The `InMemoryDocumentStore` implements the `DocumentStore` trait, allowing it to be used as a primary document store in Nova applications. It uses `RwLock` to manage concurrent access to
16/// the underlying data and index structures, making it suitable for asynchronous contexts. This adapter is ideal for testing and local development but is not recommended for production use due to its lack of persistence and scalability.
17/// The implementation provides basic CRUD operations for documents, as well as index management functions that allow you to create and list indexes for collections. The `get`, `upsert`, and `delete` methods operate on a nested `HashMap` structure where the first level key is the collection name and the second level key is the document ID.
18/// The `create_index` and `list_indexes` methods manage a separate `HashMap` that tracks indexes for each collection, allowing you to simulate index creation and retrieval in an in-memory context. Overall, this adapter serves as a simple and effective way to test NoSQL document store functionality without needing to set up an actual database backend.
19/// Note that since this is an in-memory store, all data will be lost when the application is restarted, and it does not support advanced features like transactions or complex queries. It is primarily intended for use in unit tests or as a lightweight option during development.
20#[async_trait]
21impl DocumentStore for InMemoryDocumentStore {
22    async fn get(&self, collection: &str, id: &str) -> Result<Option<JsonValue>, NoSqlError> {
23        let data = self.data.read().await;
24        Ok(data.get(collection).and_then(|c| c.get(id)).cloned())
25    }
26
27    async fn upsert(&self, collection: &str, id: &str, doc: JsonValue) -> Result<(), NoSqlError> {
28        let mut data = self.data.write().await;
29        data.entry(collection.to_string())
30            .or_default()
31            .insert(id.to_string(), doc);
32        Ok(())
33    }
34
35    async fn delete(&self, collection: &str, id: &str) -> Result<(), NoSqlError> {
36        let mut data = self.data.write().await;
37        if let Some(col) = data.get_mut(collection) {
38            col.remove(id);
39        }
40        Ok(())
41    }
42
43    async fn create_index(&self, collection: &str, index: NoSqlIndex) -> Result<(), NoSqlError> {
44        let mut indexes = self.indexes.write().await;
45        indexes
46            .entry(collection.to_string())
47            .or_default()
48            .push(index);
49        Ok(())
50    }
51
52    async fn list_indexes(&self, collection: &str) -> Result<Vec<NoSqlIndex>, NoSqlError> {
53        let indexes = self.indexes.read().await;
54        Ok(indexes.get(collection).cloned().unwrap_or_default())
55    }
56}
