5/// Adapter abstraction for NoSQL primary document stores.
6#[async_trait]
7pub trait DocumentStore: Send + Sync {
8    /// Get a document by collection and id, returning `None` if not found.
9    async fn get(&self, collection: &str, id: &str) -> Result<Option<JsonValue>, NoSqlError>;
10    /// Upsert a document into the store.
11    async fn upsert(&self, collection: &str, id: &str, doc: JsonValue) -> Result<(), NoSqlError>;
12    /// Delete a document from the store.
13    async fn delete(&self, collection: &str, id: &str) -> Result<(), NoSqlError>;
14    /// Index management helpers proxying to the primary adapter.
15    async fn create_index(&self, collection: &str, index: NoSqlIndex) -> Result<(), NoSqlError>;
16    /// List indexes for a collection.
17    async fn list_indexes(&self, collection: &str) -> Result<Vec<NoSqlIndex>, NoSqlError>;
18}

20/// Optional cache adapter for document payloads.
