1//! Small utility for composing OpenAPI fragments contributed by multiple crates.
2//!
3//! Crates can register an `OpenApiHook` via the `inventory` crate. The
4//! `build_openapi_document` function iterates the collected hooks and merges
5//! their JSON fragments into a single OpenAPI document.
6use inventory;
7use serde_json::{Map, Value, json};
8
9/// A single OpenAPI fragment provider.
10///
11/// `name` is a human-friendly identifier used for debugging; `provider` must
12/// return a JSON `Value` containing a valid OpenAPI fragment (typically a
13/// partial document with `paths` and/or `components`).
14pub struct OpenApiHook {
15    pub name: &'static str,
16    pub provider: fn() -> Value,
17}
18
19inventory::collect!(OpenApiHook);
20
21/// Merge `overlay` into `base` recursively. Objects are merged by key,
22/// other JSON values replace the base value.
23fn merge_json(base: &mut Value, overlay: Value) {
24    match (base, overlay) {
25        (Value::Object(base_map), Value::Object(overlay_map)) => {
26            for (key, value) in overlay_map {
27                match base_map.get_mut(&key) {
28                    Some(existing) => merge_json(existing, value),
29                    None => {
30                        base_map.insert(key, value);
31                    }
32                }
33            }
34        }
35        (base_slot, overlay_value) => {
36            *base_slot = overlay_value;
37        }
38    }
39}
40
41pub fn build_openapi_document(service_name: &str) -> Value {
42    let mut doc = json!({
43        "openapi": "3.1.0",
44        "info": {
45            "title": format!("{service_name} API"),
46            "version": "0.1.0"
47        },
48        "paths": Value::Object(Map::new())
49    });
50
51    for hook in inventory::iter::<OpenApiHook> {
52        let fragment = (hook.provider)();
53        merge_json(&mut doc, fragment);
54    }
55
56    doc
57}
